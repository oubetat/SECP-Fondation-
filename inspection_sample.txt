  public authorizeCalibration(cert: CalibrationCertificate, actor: string): void {

      throw new Error(
        `AI_GOVERNANCE_VIOLATION: AI agent '${actor}' is unauthorized to sign or authorize Calibration Certificates. Human validation required.`
      );
